cmake_minimum_required(VERSION 3.24)
project(Lumen VERSION 5.0.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(LUMEN_WARNINGS "Warnings stricts" ON)
option(LUMEN_HARDENING "Durcissement compilateur/linker" ON)
option(LUMEN_WERROR "Traiter les warnings comme erreurs" OFF)
option(LUMEN_USE_LLAMA "Activer llama.cpp" ON)

if(MSVC)
  add_compile_options(/utf-8) # sources en UTF-8 : prompts et textes accentués corrects
endif()

if(LUMEN_WARNINGS)
  if(MSVC)
    add_compile_options(/W4 /permissive-)
  else()
    add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wformat=2 -Wconversion -Wstrict-prototypes)
  endif()
endif()
if(LUMEN_WERROR)
  set(CMAKE_COMPILE_WARNING_AS_ERROR ON)
endif()
if(LUMEN_HARDENING AND NOT MSVC)
  add_compile_options(-fstack-protector-strong -D_FORTIFY_SOURCE=2)
  add_link_options(-Wl,-z,relro,-z,now)
endif()

set(SOURCES src/main.c src/search.c src/image.c src/rag.c src/server.c src/gui.c src/session.c)
if(LUMEN_USE_LLAMA)
  list(APPEND SOURCES src/llm.c)
else()
  list(APPEND SOURCES src/llm_stub.c)
endif()
add_executable(Lumen ${SOURCES})
target_include_directories(Lumen PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

target_compile_definitions(Lumen PRIVATE LUMEN_VERSION=\"${PROJECT_VERSION}\")
find_package(Threads REQUIRED)
# Prefer the shared libcurl when available. Some environments expose a static
# /usr/local/lib/libcurl.a without its transitive dependencies, which makes a
# naive CURL::libcurl link fail at the final linker step.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(CURL_PC QUIET libcurl)
endif()
if(CURL_PC_FOUND)
  target_include_directories(Lumen PRIVATE ${CURL_PC_INCLUDE_DIRS})
  target_link_directories(Lumen PRIVATE ${CURL_PC_LIBRARY_DIRS})
  target_link_libraries(Lumen PRIVATE ${CURL_PC_LIBRARIES} Threads::Threads)
else()
  find_package(CURL REQUIRED)
  target_link_libraries(Lumen PRIVATE CURL::libcurl Threads::Threads)
endif()
if(NOT WIN32)
  target_link_libraries(Lumen PRIVATE m)
endif()
# cJSON : paquet système recommandé, avec repli vers le header de compatibilité fourni.
find_package(cJSON CONFIG QUIET)
if(cJSON_FOUND)
  target_link_libraries(Lumen PRIVATE cjson)
else()
  find_path(CJSON_INCLUDE_DIR cJSON.h PATHS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/cjson /usr/include/cjson /usr/local/include/cjson)
  find_library(CJSON_LIBRARY NAMES cjson libcjson.so libcjson.so.1 PATHS /usr/lib /usr/lib/x86_64-linux-gnu /usr/local/lib)
  if(NOT CJSON_INCLUDE_DIR OR NOT CJSON_LIBRARY)
    message(FATAL_ERROR "cJSON introuvable. Installez cJSON ou fournissez CJSON_INCLUDE_DIR/CJSON_LIBRARY.")
  endif()
  target_include_directories(Lumen PRIVATE ${CJSON_INCLUDE_DIR})
  target_link_libraries(Lumen PRIVATE ${CJSON_LIBRARY})
endif()

if(WIN32)
  target_link_libraries(Lumen PRIVATE ws2_32 user32 gdi32)
endif()

if(LUMEN_USE_LLAMA)
  set(LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp" CACHE PATH "Chemin vers llama.cpp")
  if(EXISTS "${LLAMA_DIR}/CMakeLists.txt")
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(${LLAMA_DIR} ${CMAKE_BINARY_DIR}/llama.cpp EXCLUDE_FROM_ALL)
    target_link_libraries(Lumen PRIVATE llama)
    target_include_directories(Lumen PRIVATE ${LLAMA_DIR}/include ${LLAMA_DIR}/ggml/include)
  else()
    message(FATAL_ERROR "llama.cpp absent: ${LLAMA_DIR}")
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS Lumen RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
message(STATUS "Lumen ${PROJECT_VERSION} | llama=${LUMEN_USE_LLAMA} | hardening=${LUMEN_HARDENING}")
